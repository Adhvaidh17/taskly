String attachmentUnavailableText(int currentProfileId) {
  final mime = attachmentMimeType ?? '';
  final kind = mime.startsWith('image/')
      ? 'image'
      : mime.startsWith('video/')
          ? 'video'
          : mime.startsWith('audio/')
              ? 'audio'
              : 'attachment';
  if (isMine(currentProfileId)) {
    return 'This $kind is not available on this device.';
  }
  final senderName = sender.name.trim().isEmpty ? 'the sender' : sender.name.trim();
  return "This $kind is not available on $senderName's device.";
}
