import '../widgets/unavailable_attachment_card.dart';

// In the attachment/image bubble builder, BEFORE any Image.file/Image.network
// or file-card code, add this first branch:
final attachmentUnavailable =
    message.isAttachmentUnavailable ||
    chatProvider.unavailableAttachmentIds.contains(message.id);

if (attachmentUnavailable) {
  return UnavailableAttachmentCard(
    message: message,
    currentProfileId: chatProvider.currentProfileId ?? 0,
  );
}

// Keep your CURRENT attachment/image card dimensions and buttons. Its tap must
// call ONLY this; do not preload the full image in build():
onTap: () async {
  try {
    await chatProvider.openMessageAttachment(message);
  } on FileSystemException {
    if (context.mounted) setState(() {});
  }
},
